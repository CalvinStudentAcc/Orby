import React, { useState } from "react"
import { Users, Lock } from "lucide-react"

import MasterDesign  from "./designs/Master"
import ClavinDesign  from "./designs/Clavin"
import KielDesign    from "./designs/Kiel"
import TomasDesign   from "./designs/Tomas"
import FrancisDesign from "./designs/Francis"

const WORKSPACES = [
  { id: "master",  name: "MASTER DESIGN",   component: MasterDesign,  icon: Lock  },
  { id: "clavin",  name: "Clavin's Design",  component: ClavinDesign,  icon: Users },
  { id: "kiel",    name: "Kiel's Design",    component: KielDesign,    icon: Users },
  { id: "tomas",   name: "Tomas's Design",   component: TomasDesign,   icon: Users },
  { id: "francis", name: "Francis's Design", component: FrancisDesign, icon: Users },
]

export default function App() {
  const [activeWorkspace, setActiveWorkspace] = useState(WORKSPACES[0].id)
  const ActiveComponent = WORKSPACES.find(w => w.id === activeWorkspace)?.component || MasterDesign

  return (
    <div className="min-h-screen relative">
      {/* Workspace Switcher — always visible at bottom */}
      <div className="fixed bottom-4 left-1/2 -translate-x-1/2 z-[9999] bg-card/90 backdrop-blur-md border border-border p-2 rounded-full shadow-2xl flex items-center gap-1 overflow-x-auto max-w-[95vw]">
        {WORKSPACES.map((workspace) => {
          const isActive = activeWorkspace === workspace.id
          const Icon = workspace.icon
          return (
            <button
              key={workspace.id}
              onClick={() => setActiveWorkspace(workspace.id)}
              className={`flex items-center gap-2 px-4 py-2 rounded-full text-sm font-medium transition-all whitespace-nowrap ${
                isActive
                  ? "bg-primary text-primary-foreground shadow-lg"
                  : "text-muted-foreground hover:bg-secondary hover:text-foreground"
              }`}
            >
              <Icon className="w-4 h-4" />
              {workspace.name}
            </button>
          )
        })}
      </div>

      <ActiveComponent />
    </div>
  )
}

import React, { useState, useEffect } from "react"
import {
  Menu,
  X,
  CheckCircle2,
  Calendar,
  BookOpen,
  DollarSign,
  Target,
  LayoutGrid,
} from "lucide-react"
import orbyIcon  from "@/imports/orbyIconNew1.png"
import orbyBrand from "@/imports/orbyBrandNew1-1.png"

const NAV_LINKS = [
  { label: "Notespace",   href: "#features" },
  { label: "Timespace",   href: "#features" },
  { label: "Taskspace",   href: "#features" },
  { label: "Budgetspace", href: "#features" },
]

export default function App() {
  const [isMenuOpen, setIsMenuOpen] = useState(false)
  const [scrolled,   setScrolled]   = useState(false)
  const [navHovered, setNavHovered] = useState(false)

  useEffect(() => {
    const handleScroll = () => setScrolled(window.scrollY > 24)
    window.addEventListener("scroll", handleScroll, { passive: true })
    return () => window.removeEventListener("scroll", handleScroll)
  }, [])

  const isExpanded = scrolled || navHovered

  return (
    <div className="min-h-screen font-sans bg-background text-foreground flex flex-col">

      {/* ── Navigation ─────────────────────────────────────────────────────── */}
      <nav
        className="fixed top-0 left-0 right-0 z-50 bg-background/60 backdrop-blur-md border-b border-border/40 transition-all duration-300"
        onMouseEnter={() => setNavHovered(true)}
        onMouseLeave={() => setNavHovered(false)}
      >
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 h-16 relative flex items-center">

          {/*
            Logo — absolutely positioned so `left` + `transform` can both
            animate via CSS, giving a true smooth slide from center → left.
          */}
          <div
            className="absolute flex items-center"
            style={{
              left:      isExpanded ? "0px" : "50%",
              transform: isExpanded ? "translateX(0)" : "translateX(-50%)",
              transition: "left 0.8s cubic-bezier(0.4,0,0.2,1), transform 0.8s cubic-bezier(0.4,0,0.2,1)",
            }}
          >
            <img
              src={orbyIcon}
              alt="Orby"
              className="h-8 w-auto object-contain"
              onError={(e) => { e.currentTarget.style.display = "none" }}
            />
          </div>

          {/* Right side: nav links + hamburger */}
          <div className="ml-auto flex items-center gap-6">
            {NAV_LINKS.map((link) => (
              <a
                key={link.label}
                href={link.href}
                className="text-sm font-medium hidden sm:block text-muted-foreground hover:text-foreground"
                style={{
                  opacity:       isExpanded ? 1 : 0,
                  pointerEvents: isExpanded ? "auto" : "none",
                  transition:    "opacity 0.8s ease",
                }}
              >
                {link.label}
              </a>
            ))}

            <button
              onClick={() => setIsMenuOpen(!isMenuOpen)}
              aria-label="Toggle menu"
              className="w-9 h-9 flex items-center justify-center rounded-full hover:bg-secondary transition-colors"
            >
              {isMenuOpen ? <X className="w-5 h-5" /> : <Menu className="w-5 h-5" />}
            </button>
          </div>
        </div>
      </nav>

      {/* ── Mobile Menu Overlay ─────────────────────────────────────────────── */}
      {isMenuOpen && (
        <div className="fixed inset-0 z-40 bg-background/95 backdrop-blur pt-20 px-6 flex flex-col gap-6 text-xl">
          {NAV_LINKS.map((link) => (
            <a
              key={link.label}
              href={link.href}
              className="hover:text-primary transition-colors"
              onClick={() => setIsMenuOpen(false)}
            >
              {link.label}
            </a>
          ))}
        </div>
      )}

      <main className="flex-1 flex flex-col">

        {/* ── Hero Section ─────────────────────────────────────────────────── */}
        <section className="galaxy-bg min-h-screen flex flex-col items-center justify-center pt-16 px-4 text-center relative">
          <div className="absolute inset-0 bg-gradient-to-b from-transparent via-background/20 to-background pointer-events-none" />

          <div className="relative z-10 max-w-4xl mx-auto flex flex-col items-center gap-8">
            <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-primary/20 text-primary-foreground border border-primary/30 text-sm font-mono">
              <span className="w-2 h-2 rounded-full bg-primary animate-pulse" />
              Workspace simplified
            </div>

            <h1 className="text-5xl md:text-7xl font-bold tracking-tight text-white">
              Your universe,{" "}
              <br className="hidden md:block" />
              <span className="text-transparent bg-clip-text bg-gradient-to-r from-primary to-accent">
                organized.
              </span>
            </h1>

            <p className="text-lg md:text-xl text-muted-foreground max-w-2xl">
              A minimalist multitasking workspace designed for students and
              individuals. No complicated setups — just the essentials you need
              to thrive.
            </p>
          </div>
        </section>

        {/* ── About Section ────────────────────────────────────────────────── */}
        <section id="about" className="py-24 px-4 bg-background">
          <div className="max-w-7xl mx-auto">
            <div className="flex flex-col md:flex-row items-center gap-16">

              {/* Text column */}
              <div className="flex-1 space-y-6">
                <h2 className="text-3xl md:text-5xl font-bold">
                  Cut the noise.
                  <br />
                  Focus on the work.
                </h2>
                <p className="text-muted-foreground text-lg leading-relaxed">
                  Most workspace apps give you a blank page and a million tools

                  project management. Everything has its place, so you can find
                  your flow state faster.
                </p>
                <ul className="space-y-3 pt-4">
                  {[
                    "Zero learning curve",
                    "Distraction-free interface",

                  ].map((item, i) => (
                    <li key={i} className="flex items-center gap-3 text-foreground">
                      <CheckCircle2 className="w-5 h-5 text-primary flex-shrink-0" />
                      {item}
                    </li>
                  ))}
                </ul>
              </div>

              {/* Brand image column */}
              <div className="flex-1 w-full">
                <div className="w-full max-w-lg mx-auto relative rounded-2xl overflow-hidden border border-border/50 bg-card/30 backdrop-blur-sm shadow-2xl p-4">
                  <div className="absolute inset-0 bg-gradient-to-tr from-primary/5 to-accent/5 pointer-events-none" />
                  <img
                    src={orbyBrand}
                    alt="Orby workspace interface showing all five tools"
                    className="w-full h-auto object-contain rounded-xl mix-blend-screen opacity-90 drop-shadow-[0_0_24px_rgba(129,140,248,0.25)]"
                    onError={(e) => { e.currentTarget.style.display = "none" }}
                  />
                </div>
              </div>

            </div>
          </div>
        </section>

        {/* ── Features Section ─────────────────────────────────────────────── */}
        <section id="features" className="py-24 px-4 bg-secondary/30 border-t border-border/50">
          <div className="max-w-7xl mx-auto">
            <div className="text-center max-w-2xl mx-auto mb-16">
              <h2 className="text-3xl md:text-5xl font-bold mb-6">
                Everything you need, nothing you don't.
              </h2>
              <p className="text-muted-foreground text-lg">
                Five core tools seamlessly integrated into one minimalist dashboard.
              </p>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">

================================================================================
NOTES FOR VS CODE / NON-FIGMA SETUP
================================================================================

1. INSTALL DEPENDENCIES
   npm install react react-dom lucide-react
   npm install -D vite @vitejs/plugin-react typescript tailwindcss @tailwindcss/vite

2. VITE CONFIG (vite.config.ts) — simplified version without Figma plugins:

   import { defineConfig } from "vite"
   import react from "@vitejs/plugin-react"
   import tailwindcss from "@tailwindcss/vite"
   import path from "node:path"

   export default defineConfig({
     plugins: [react(), tailwindcss()],
     resolve: {
       alias: { "@": path.resolve(__dirname, "./src") },
     },
   })

3. IMAGE IMPORTS
   In Vite, PNG/JPG files in src/ must be imported as ES modules.
   The @/imports/ alias points to src/imports/.
   Place orbyIconNew1.png and orbyBrandNew1-1.png in src/imports/.

4. TAILWIND CSS v4
   src/index.css uses Tailwind v4 syntax (@import 'tailwindcss').
   No tailwind.config.js is needed with the @tailwindcss/vite plugin.

5. PLACEHOLDER DESIGNS (Clavin, Kiel, Tomas, Francis)
   Each file in src/designs/<Name>/index.tsx is a separate teammate workspace.
   They can be copies of Master/index.tsx until each teammate builds their own.
   The workspace switcher in App.tsx is only for internal review — remove it
   when shipping the final production site.
